import React, { useState } from 'react';

export default function App() {
  const [hoveredFeature, setHoveredFeature] = useState(null);
  const [hoveredPrimer, setHoveredPrimer] = useState(null);
  const [hoveredEnzyme, setHoveredEnzyme] = useState(null);

  // Modern sans-serif monospace font stack
  const monoFont = 'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, "Liberation Mono", monospace';
  // Fast, non-linear smooth spring easing for UI transitions
  const springAnim = 'all 0.3s cubic-bezier(0.16, 1, 0.3, 1)';

  // Core Data
  const sequence = "TACGAATTCGCCACCATGGCCATGAAGCTTGAGCTCGGATCCTCTAGAGCT";
  const features = [
    { id: 'CDS', name: 'CDS', start: 15, end: 51, color: '#60A5FA', yOffset: 14 },
    { id: 'Kozak', name: 'Kozak', start: 9, end: 19, color: '#FBBF24', yOffset: 26 }
  ];

  const enzymes = [
    { id: 'EcoRI', name: 'EcoRI', cutIndex: 4, seq: 'GAATTC', compSeq: 'CTTAAG', topCut: 1, botCut: 5, isUnique: false },
    { id: 'XbaI', name: 'XbaI', cutIndex: 43, seq: 'TCTAGA', compSeq: 'AGATCT', topCut: 1, botCut: 5, isUnique: true }
  ];

  const primers = [
    {
      id: 'Fwd', name: 'Fwd Primer', type: 'fwd',
      matchStart: 0, matchEnd: 10, mismatchStr: 'ctga', matchStr: 'TACGAATTCG',
      color: '#166534',
      getPoints: (sx, cw, seqY) => {
        const misY = seqY - 50;
        const matchY = seqY - 30;
        return [
          [sx - 4 * cw, misY],
          [sx - 0.5 * cw, misY],
          [sx + 0 * cw, matchY],
          [sx + 10 * cw, matchY]
        ];
      }
    },
    {
      id: 'Rev', name: 'Rev Primer', type: 'rev',
      matchStart: 31, matchEnd: 42, mismatchStr: 'cctg', matchStr: 'GGATCCGAGCT',
      color: '#166534',
      getPoints: (sx, cw, seqY) => {
        const misY = seqY + 41; 
        const matchY = seqY + 26; 
        return [
          [sx + 46 * cw, misY],
          [sx + 42.5 * cw, misY],
          [sx + 42 * cw, matchY],
          [sx + 31 * cw, matchY]
        ];
      }
    }
  ];

  // Layout Metrics
  const cw = 14; 
  const startX = 150; 
  const seqY = 250; 
  const svgWidth = startX + sequence.length * cw + 150;
  const svgHeight = 500;
  const bgColor = "#fdfbf7"; 

  // Render Helpers
  const renderFeatures = () => {
    return features.map(f => {
      const y = seqY + f.yOffset;
      const x1 = startX + f.start * cw;
      const x2 = startX + f.end * cw;
      const isHovered = hoveredFeature === f.id;

      return (
        <g 
          key={f.id} 
          onMouseEnter={() => setHoveredFeature(f.id)} 
          onMouseLeave={() => setHoveredFeature(null)}
          className="cursor-pointer"
        >
          {/* Animated Highlight Rect */}
          <rect 
            x={x1} 
            y={isHovered ? seqY - 18 : y} 
            width={x2 - x1} 
            height={isHovered ? y - (seqY - 18) : 0} 
            fill={f.color} 
            fillOpacity={0.15} 
            style={{ transition: springAnim }}
          />
          <line x1={x1} x2={x2} y1={y} y2={y} stroke={f.color} strokeWidth="5" />
          <text x={x1 - 8} y={y + 4} fill={f.color} fontSize="12px" fontFamily="sans-serif" fontWeight="600" textAnchor="end">
            {f.name}
          </text>
          <line x1={x1} x2={x2} y1={y} y2={y} stroke="transparent" strokeWidth="20" />
        </g>
      );
    });
  };

  const renderPrimers = () => {
    return primers.map(p => {
      const isHovered = hoveredPrimer === p.id;
      const pts = p.getPoints(startX, cw, seqY);
      const pathStr = `M ${pts[0][0]} ${pts[0][1]} L ${pts[1][0]} ${pts[1][1]} L ${pts[2][0]} ${pts[2][1]} L ${pts[3][0]} ${pts[3][1]}`;
      
      const isFwd = p.type === 'fwd';
      const arrowTipX = pts[3][0];
      const arrowTipY = pts[3][1];
      const arrowBackX = isFwd ? arrowTipX - 0.5 * cw : arrowTipX + 0.5 * cw;
      const arrowBackY = isFwd ? arrowTipY - 5 : arrowTipY + 5;
      const arrowPath = `M ${arrowTipX} ${arrowTipY} L ${arrowBackX} ${arrowBackY}`;

      // Dynamic expansion height for morphing paths
      const maxExp = 26;
      const currentHExp = isHovered ? maxExp : 0;
      
      // Structure as a closed 8-point path for perfect transition morphing.
      const hoverBoxPath = isFwd
        ? `M ${pts[0][0]} ${pts[0][1]} L ${pts[1][0]} ${pts[1][1]} L ${pts[2][0]} ${pts[2][1]} L ${pts[3][0]} ${pts[3][1]} L ${pts[3][0]} ${pts[3][1] - currentHExp} L ${pts[2][0]} ${pts[2][1] - currentHExp} L ${pts[1][0]} ${pts[1][1] - currentHExp} L ${pts[0][0]} ${pts[0][1] - currentHExp} Z`
        : `M ${pts[0][0]} ${pts[0][1]} L ${pts[1][0]} ${pts[1][1]} L ${pts[2][0]} ${pts[2][1]} L ${pts[3][0]} ${pts[3][1]} L ${pts[3][0]} ${pts[3][1] + currentHExp} L ${pts[2][0]} ${pts[2][1] + currentHExp} L ${pts[1][0]} ${pts[1][1] + currentHExp} L ${pts[0][0]} ${pts[0][1] + currentHExp} Z`;

      return (
        <g 
          key={p.id} 
          onMouseEnter={() => setHoveredPrimer(p.id)} 
          onMouseLeave={() => setHoveredPrimer(null)}
          className="cursor-pointer"
        >
          {/* 1. OPAQUE MASK: Completely blocks background (e.g. Enzyme lines) before applying translucent color */}
          <path 
            d={hoverBoxPath} 
            fill={bgColor} 
            style={{ transition: springAnim }} 
          />
          {/* 2. TRANSLUCENT COLOR: The actual colored hover area */}
          <path 
            d={hoverBoxPath} 
            fill={p.color} 
            fillOpacity={0.1} 
            style={{ transition: springAnim }} 
          />
          
          {/* Sequence inner text (always positioned at max expanded spot, fades in/out) */}
          <text 
            fill={p.color} fontSize="14px" fontFamily={monoFont} fontWeight="bold"
            style={{ opacity: isHovered ? 1 : 0, transition: 'opacity 0.2s ease-in-out', pointerEvents: 'none' }}
          >
            {isFwd ? (
              <>
                {p.mismatchStr.split('').map((c, i) => (
                  <tspan key={`mis-${i}`} x={pts[0][0] + i * cw + cw / 2} y={pts[0][1] - 8} textAnchor="middle">{c}</tspan>
                ))}
                {p.matchStr.split('').map((c, i) => (
                  <tspan key={`mat-${i}`} x={pts[2][0] + i * cw + cw / 2} y={pts[2][1] - 8} textAnchor="middle">{c}</tspan>
                ))}
              </>
            ) : (
              <>
                {p.mismatchStr.split('').map((c, i) => (
                  <tspan key={`mis-${i}`} x={pts[0][0] - i * cw - cw / 2} y={pts[0][1] + 18} textAnchor="middle">{c}</tspan>
                ))}
                {p.matchStr.split('').map((c, i) => (
                  <tspan key={`mat-${i}`} x={pts[2][0] - i * cw - cw / 2} y={pts[2][1] + 18} textAnchor="middle">{c}</tspan>
                ))}
              </>
            )}
          </text>

          {/* Halo mask to visually cut elements below this primer's main lines */}
          <path d={pathStr} fill="none" stroke={bgColor} strokeWidth="6" strokeLinejoin="round" />
          <path d={arrowPath} fill="none" stroke={bgColor} strokeWidth="6" strokeLinecap="round" strokeLinejoin="round" />

          {/* Main Path & Arrow Head */}
          <path d={pathStr} fill="none" stroke={p.color} strokeWidth="2.5" />
          <path d={arrowPath} fill="none" stroke={p.color} strokeWidth="2.5" strokeLinecap="round" />
          
          {/* Fade out primer name on hover, placed above (Fwd) or below (Rev) the arrow */}
          <text 
            x={pts[0][0]} 
            y={isFwd ? pts[0][1] - 8 : pts[0][1] + 20} 
            fill={p.color} fontSize="12px" fontFamily="sans-serif" fontWeight="600" textAnchor={isFwd ? "start" : "end"}
            style={{ 
              opacity: isHovered ? 0 : 1, 
              transition: springAnim,
              pointerEvents: 'none'
            }}
          >
            {p.name}
          </text>

          <path d={pathStr} fill="none" stroke="transparent" strokeWidth="20" />
        </g>
      );
    });
  };

  const renderEnzymes = (isTopLayer) => {
    return enzymes.map(e => {
      const isHovered = hoveredEnzyme === e.id;
      
      if (isTopLayer && !isHovered) return null;
      if (!isTopLayer && isHovered) return null;

      const cutX = startX + e.cutIndex * cw;
      const color = isHovered ? '#2563EB' : '#333';
      const strokeW = isHovered ? "2" : "1";
      // Font weight never changes on hover, only depends on uniqueness
      const fontW = e.isUnique ? "bold" : "normal"; 
      const textDeco = "none";
      
      let highestObstacleY = seqY - 40; 
      const topPrimers = primers.filter(p => p.type === 'fwd');
      for (const p of topPrimers) {
        const pts = p.getPoints(startX, cw, seqY);
        if (cutX >= pts[0][0] - cw && cutX <= pts[3][0] + cw) {
          highestObstacleY = Math.min(highestObstacleY, pts[0][1] - 26);
        }
      }
      
      const yTop = highestObstacleY - 25; 
      const yBottom = isHovered ? (seqY - 25) : (seqY - 15);

      return (
        <g key={e.id}>
          {isHovered && (
            <line x1={cutX} x2={cutX} y1={yTop} y2={yBottom} stroke={bgColor} strokeWidth="6" strokeLinecap="square" />
          )}

          <line x1={cutX} x2={cutX} y1={yTop} y2={yBottom} stroke={color} strokeWidth={strokeW} />
          
          {/* Hover trigger is purely isolated to the text itself */}
          <text 
            x={cutX + 6} y={yTop + 5} fill={color} fontSize="14px" 
            fontFamily="sans-serif" fontWeight={fontW} textDecoration={textDeco}
            onMouseEnter={() => setHoveredEnzyme(e.id)} 
            onMouseLeave={() => setHoveredEnzyme(null)}
            className="cursor-pointer"
          >
            {e.name}
          </text>
        </g>
      );
    });
  };

  const renderTooltips = () => {
    // Only render the hovered tooltip instantly without transitions
    if (!hoveredEnzyme) return null;
    
    const e = enzymes.find(x => x.id === hoveredEnzyme);
    const startIndex = e.cutIndex - e.topCut;
    const len = e.seq.length;
    
    const leftX = startX + startIndex * cw - 8;
    const rightX = startX + (startIndex + len) * cw + 8;
    const ttWidth = rightX - leftX;
    const ttHeight = 60; 
    const ttY = seqY - 26;
    
    const cutX = startX + e.cutIndex * cw;
    const botCutX = cutX + (e.botCut - e.topCut) * cw;

    return (
      <g key={`tooltip-${e.id}`}>
        <rect 
          x={leftX} y={ttY} width={ttWidth} height={ttHeight} rx={8} 
          fill="#FFFFFF" stroke="#2563EB" strokeWidth="2"
        />
        {/* Adjusted mask width from 10 to 6 pixels to perfectly match the background halo strokeWidth="6" */}
        <line x1={cutX - 3} x2={cutX + 3} y1={ttY} y2={ttY} stroke="#FFFFFF" strokeWidth="3" />
        <path 
          d={`M ${cutX} ${ttY - 2} L ${cutX} ${seqY + 3} L ${botCutX} ${seqY + 3} L ${botCutX} ${seqY + 28}`} 
          fill="none" stroke="#2563EB" strokeWidth="2" strokeLinejoin="round"
        />
        <text y={seqY} fontFamily={monoFont} fontSize="18px" fontWeight="bold" fill="#1f2937">
          {e.seq.split('').map((c, i) => (
            <tspan key={`t-${i}`} x={startX + (startIndex + i) * cw + cw / 2} textAnchor="middle">{c}</tspan>
          ))}
        </text>
        <text y={seqY + 20} fontFamily={monoFont} fontSize="18px" fontWeight="bold" fill="#6B7280">
          {e.compSeq.split('').map((c, i) => (
            <tspan key={`b-${i}`} x={startX + (startIndex + i) * cw + cw / 2} textAnchor="middle">{c}</tspan>
          ))}
        </text>
      </g>
    );
  };

  return (
    <div className="w-full min-h-screen bg-[#fdfbf7] flex justify-center items-start pt-16 overflow-x-auto select-none">
      <div style={{ width: svgWidth }}>
        <svg width="100%" height={svgHeight} className="block">
          {/* Layer 1: Bottom features highlights and lines */}
          {renderFeatures()}
          
          {/* Layer 2: Unhovered enzymes (Masked/cut by Primers) */}
          {renderEnzymes(false)}

          {/* Layer 3: Primers (Has solid base to cut Unhovered enzymes underneath during hover) */}
          {renderPrimers()}

          {/* Layer 4: Hovered enzyme (Has halo to cut Primers underneath) */}
          {renderEnzymes(true)}

          {/* Layer 5: Sequence Text */}
          <text 
            y={seqY} fontFamily={monoFont} fontSize="18px" fontWeight="bold" 
            fill="#1f2937" className="cursor-text select-text" style={{ userSelect: 'text' }}
          >
            {sequence.split('').map((char, i) => (
              <tspan key={i} x={startX + i * cw + cw / 2} textAnchor="middle">{char}</tspan>
            ))}
          </text>

          {/* Layer 6: Hovered Tooltips */}
          {renderTooltips()}
        </svg>
      </div>
    </div>
  );
}
